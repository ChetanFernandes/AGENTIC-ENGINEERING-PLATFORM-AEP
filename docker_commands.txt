# Validate configuration
    docker compose config

# Build and start services
    docker compose up -d --build

# Check running containers
    docker compose ps

# Stop services
    docker compose down

# View live logs
    docker compose logs -f

# Logs from a specific container
    docker logs -f <container_name>

# API logs
    docker compose logs -f api

# Streamlit UI logs
    docker compose logs -f ui

# Show the last 100 log lines
    docker compose logs --tail=100 -f api

# Show logs with timestamps
    docker compose logs -f -t

# Check log files exist
    docker compose exec api sh -c "pwd && ls -l logs/"

# Watch your application's JSON log file
    docker compose exec api sh -c "tail -n 100 -f logs/log.txt"

# Stops all running containers but keeps them for later restart
    docker compose stop	

# removes containers and the Compose network. Your named volumes are preserved by default.
    docker compose down	Stops 

# Restarts previously stopped containers.
    docker compose start

# docker compose exec postgres psql -U postgres -d postgres
    a. docker compose exec postgres — targets the postgres service defined in your docker-compose.yml.
    b. psql -U postgres -d postgres — opens the PostgreSQL command-line client using the postgres user and connects to the default postgres database.
